cmake_minimum_required(VERSION 3.21)
project(GestaoEquipamentos VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Sql)
qt_standard_project_setup()
include(cmake/GestaoEquipamentosDependencies.cmake)
add_library(notebook_core src/storage/LocalDatabase.cpp src/storage/NotebookStore.cpp src/domain/HistoryPeriod.cpp src/domain/EventCatalog.cpp src/application/HistoryQueryService.cpp
 src/security/Crypto.cpp src/security/PasswordHasher.cpp src/security/SecretVault.cpp src/security/Totp.cpp src/security/RecoveryCodes.cpp
 src/application/AuthService.cpp src/application/AccountService.cpp src/application/AdminProtocol.cpp src/application/DemoSeed.cpp
 src/application/RecoveryService.cpp src/application/LocationService.cpp src/application/PeopleService.cpp)
target_include_directories(notebook_core PUBLIC src)
target_link_libraries(notebook_core PUBLIC Qt6::Core Qt6::Sql sodium::sodium PRIVATE gestaoequipamentos_keychain)
add_library(qrcodegen STATIC third_party/qrcodegen/qrcodegen.cpp)
target_include_directories(qrcodegen PUBLIC third_party/qrcodegen)
set(GESTAOEQUIPAMENTOS_UI_SOURCES src/ui/MainWindow.cpp src/ui/LogsPage.cpp src/ui/HistoryFormat.cpp src/ui/AccessDialogs.cpp src/ui/AccountDialogs.cpp src/ui/AccountsPage.cpp
 src/ui/QrCodeView.cpp src/ui/RecoveryDialogs.cpp src/ui/LocationsPage.cpp src/ui/PeoplePage.cpp
 src/ui/PersonPickerDialog.cpp src/ui/PersonDialog.cpp src/ui/LocationDialog.cpp src/ui/QuickPickField.cpp)
qt_add_executable(GestaoEquipamentos src/main.cpp ${GESTAOEQUIPAMENTOS_UI_SOURCES} resources/resources.qrc)
set_target_properties(GestaoEquipamentos PROPERTIES AUTORCC ON WIN32_EXECUTABLE ON MACOSX_BUNDLE ON)
target_link_libraries(GestaoEquipamentos PRIVATE notebook_core qrcodegen Qt6::Widgets)
gestaoequipamentos_copy_runtime_dlls(GestaoEquipamentos)
qt_add_executable(GestaoEquipamentosAdmin src/admin/admin_main.cpp)
set_target_properties(GestaoEquipamentosAdmin PROPERTIES WIN32_EXECUTABLE OFF MACOSX_BUNDLE OFF)
target_link_libraries(GestaoEquipamentosAdmin PRIVATE notebook_core)
gestaoequipamentos_copy_runtime_dlls(GestaoEquipamentosAdmin)
include(CTest)
if(BUILD_TESTING)

 set_tests_properties(movement_tests PROPERTIES TIMEOUT 120)
 qt_add_executable(catalog_tests tests/CatalogTests.cpp)

 add_test(NAME catalog_tests COMMAND catalog_tests)
 set_tests_properties(catalog_tests PROPERTIES TIMEOUT 120)
 qt_add_executable(auth_tests tests/AuthTests.cpp)
 target_link_libraries(auth_tests PRIVATE notebook_core Qt6::Test)
 add_test(NAME auth_tests COMMAND auth_tests)
 set_tests_properties(auth_tests PROPERTIES TIMEOUT 120)
 qt_add_executable(account_tests tests/AccountTests.cpp)

 qt_add_executable(recovery_tests tests/RecoveryTests.cpp)
 target_link_libraries(recovery_tests PRIVATE notebook_core Qt6::Test)
 add_test(NAME recovery_tests COMMAND recovery_tests)

 qt_add_executable(vault_tests tests/VaultTests.cpp)
 target_link_libraries(vault_tests PRIVATE notebook_core Qt6::Test)

 set_tests_properties(vault_tests PROPERTIES TIMEOUT 60 RESOURCE_LOCK gestaoequipamentos_vault_test SKIP_REGULAR_EXPRESSION "SKIP   : VaultTests::")

 target_link_libraries(admin_cli_tests PRIVATE notebook_core Qt6::Test)

 target_link_libraries(e2e_tests PRIVATE notebook_core Qt6::Test)

 add_test(NAME ui_tests COMMAND ui_tests -o -,txt)

 if(WIN32)
  set_property(TEST ui_tests APPEND PROPERTY ENVIRONMENT "QT_QPA_FONTDIR=$ENV{WINDIR}/Fonts")
 endif()
 foreach(test_target history_query_tests store_tests movement_tests catalog_tests auth_tests account_tests admin_tests recovery_tests vault_tests admin_cli_tests e2e_tests ui_tests)
  gestaoequipamentos_copy_runtime_dlls(${test_target})
 endforeach()
 find_package(Python3 COMPONENTS Interpreter)
 if(Python3_Interpreter_FOUND)
  add_test(NAME admin_script_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_admin_account.py)
 endif()
endif()
